function toggleFullscreen() { const container = document.getElementById('boardContainer'); const gameInfoCol = document.getElementById('gameInfoColumn'); const btn = document.getElementById('fullscreenExitBtn'); if (container.classList.contains('fullscreen')) { // Exit fullscreen container.classList.remove('fullscreen'); document.body.classList.remove('fullscreen-active'); gameInfoCol.style.display = 'block'; btn.style.display = 'none'; } else { // Enter fullscreen container.classList.add('fullscreen'); document.body.classList.add('fullscreen-active'); gameInfoCol.style.display = 'none'; btn.style.display = 'block'; } // Redraw board to fit new size with longer delay if (board) { setTimeout(() => { board.resize(); }, 300); } }
